class ReActEngine:
    def __init__(self, ...):
        self.loop_detector = LoopDetector(
            window_size=5,
            similarity_threshold=0.8,
        )
    
    def run(self, prompt: str, ...) -> str:
        messages = [...]
        self.loop_detector.reset()
        
        iteration = 0
        while iteration < MAX_ITERATIONS:  # 保留上限作为保险
            response = self._call_llm(messages)
            
            if has_final_answer(response):
                return response
            
            # 添加到循环检测器
            self.loop_detector.add_turn(
                output=response.content,
                tool_calls=[t.name for t in response.tools],
                error=response.error if hasattr(response, 'error') else None,
                thought=response.reasoning if hasattr(response, 'reasoning') else None,
            )
            
            # 检查循环
            loop_result = self.loop_detector.check()
            if loop_result.is_loop:
                return self._handle_loop(loop_result, messages)
            
            tool_results = self._execute_tools(response)
            messages.append(tool_results)
            iteration += 1
        
        raise MaxIterationsExceeded()
    
    def _handle_loop(self, loop_result: LoopDetectionResult, messages: list) -> str:
        """处理检测到的循环"""
        logger.warning(f"检测到循环: {loop_result.reason}")
        logger.warning(f"周期长度: {loop_result.loop_length}, 置信度: {loop_result.confidence}")
        
        # 策略 1: 向用户报告并请求干预
        error_msg = (
            f"检测到任务陷入循环（{loop_result.reason}），"
            f"周期长度: {loop_result.loop_length}。"
            f"建议：请检查任务需求或尝试不同的方法。"
        )
        
        # 策略 2: 尝试打破循环
        # - 注入"换一个思路"提示
        # - 调整执行策略（放宽权限）
        # - 切换到不同的工具集
        
        raise LoopDetectedError(error_msg, loop_result=loop_result)

from xenon.engine.loop_detector import LoopDetector, LoopDetectionResult

class ReActEngine(BaseEngine):

### Step 2: 在主循环中集成

```python
def run(
    self,
    prompt: str,
    context: dict[str, Any] | None = None,
    tracker: ToolTracker | None = None,
) -> str:
    """运行 ReAct 循环"""
    
    # 重置循环检测器
    self.loop_detector.reset()
    
    messages = self._build_initial_messages(prompt, context)
    iteration = 0
    max_iterations = 20  # 保留作为硬上限
    
    while iteration < max_iterations:
        # 调用 LLM
        response = self._call_llm_for_phase("reason_act", messages)
        
        # 解析响应
        parsed = self._parse_response(response)
        
        # 添加到循环检测器
        self.loop_detector.add_turn(
            output=response,
            tool_calls=[action.get("action") for action in parsed.get("actions", [])],
            error=self._get_last_error(),  # 如果有的话
            thought=parsed.get("thought", ""),
        )
        
        # 检查循环
        loop_result = self.loop_detector.check()
        if loop_result.is_loop:
            return self._handle_loop_detection(loop_result, messages, iteration)
        
        # 如果有 final_answer，返回
        if parsed.get("final_answer"):
            return parsed["final_answer"]
        
        # 执行工具
        tool_results = self._execute_actions(parsed.get("actions", []), tracker)
        
        # 添加到 messages
        messages.append({
            "role": "assistant",
            "content": response,
        })
        messages.append({
            "role": "user",
            "content": self._format_tool_results(tool_results),
        })
        
        iteration += 1
    
    # 达到硬上限
    raise MaxIterationsExceeded(
        f"ReAct 达到最大迭代次数 ({max_iterations})，"
        f"循环检测器统计: {self.loop_detector.stats()}"
    )
```

### Step 3: 实现循环处理策略

```python
def _handle_loop_detection(
    self,
    loop_result: LoopDetectionResult,
    messages: list,
    current_iteration: int,
) -> str:
    """
    处理检测到的循环。
    
    提供多种策略：
    1. 报告并终止
    2. 尝试打破循环（注入提示）
    3. 调整执行策略
    """
    logger.warning(
        f"🔄 检测到循环 (轮次 {current_iteration}): {loop_result.reason}"
    )
    logger.warning(
        f"   周期长度: {loop_result.loop_length}, "
        f"置信度: {loop_result.confidence:.2f}, "
        f"相似轮次: {loop_result.similar_turns}"
    )
    
    # 策略 1: 低置信度 - 注入打破循环的提示
    if loop_result.confidence < 0.9 and current_iteration < 15:
        logger.info("尝试注入提示打破循环...")
        
        break_loop_hint = (
            "\n\n[系统提示] 检测到可能的循环模式。"
            "建议：\n"
            "1. 检查之前尝试的方法是否有效\n"
            "2. 如果某个方法失败了多次，尝试完全不同的方法\n"
            "3. 考虑是否需要调整执行策略或权限\n"
            "4. 如果任务确实无法完成，请在 final_answer 中说明原因\n"
        )
        
        messages.append({
            "role": "user",
            "content": break_loop_hint,
        })
        
        # 继续执行，但只给一次机会
        return self._continue_with_hint(messages)
    
    # 策略 2: 高置信度 - 终止并报告
    error_msg = (
        f"任务陷入循环无法继续：\n"
        f"  原因: {loop_result.reason}\n"
        f"  周期长度: {loop_result.loop_length}\n"
        f"  相似轮次: {loop_result.similar_turns}\n"
        f"  置信度: {loop_result.confidence:.2f}\n"
        f"\n"
        f"建议：\n"
        f"  - 检查任务需求是否明确\n"
        f"  - 尝试分解任务为更小的步骤\n"
        f"  - 检查执行策略是否过于严格\n"
    )
    
    raise LoopDetectedError(error_msg, loop_result=loop_result)

def _continue_with_hint(self, messages: list) -> str:
    """注入提示后继续执行一轮"""
    response = self._call_llm_for_phase("reason_act", messages)
    parsed = self._parse_response(response)
    
    if parsed.get("final_answer"):
        return parsed["final_answer"]
    
    # 如果还是没有 final_answer，终止
    raise LoopDetectedError(
        "注入打破循环提示后仍未得到 final_answer，任务终止。"
    )
```

class LoopDetectedError(Exception):
    """检测到循环时抛出"""
